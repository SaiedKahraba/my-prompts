<script>

// الوظائف الأساسية للموقع

function init() { renderSidebar(); renderHome(); }


function renderSidebar() {

    const container = document.getElementById('sidebarCategories');

    container.innerHTML = '';

    rawData.forEach(cat => {

        const btn = document.createElement('button');

        btn.className = "sidebar-item w-full text-right px-3 py-1.5 rounded-md text-sm transition-colors flex items-center gap-2 justify-end";

        btn.id = `side-${cat.id}`;

        btn.onclick = () => showCategory(cat.id);

        btn.innerHTML = `<span class="truncate">${cat.title}</span> <span>${cat.icon}</span>`;

        container.appendChild(btn);

    });

}


function renderHome() {

    const grid = document.getElementById('categoriesGrid');

    grid.innerHTML = '';

    document.getElementById('homeView').classList.remove('hidden');

    document.getElementById('detailView').classList.add('hidden');

    document.querySelectorAll('.sidebar-item').forEach(b => b.classList.remove('active'));

    document.getElementById('btn-home').classList.add('active');

    rawData.forEach(cat => {

        const card = document.createElement('div');

        card.className = "category-card notion-shadow";

        card.onclick = () => showCategory(cat.id);

        card.innerHTML = `<div class="icon-box" style="background-color: ${cat.color}">${cat.icon}</div><div class="font-bold text-[13px] leading-tight text-gray-700">${cat.title}</div><div class="text-[10px] text-gray-400 mt-2">${cat.items.length} برومبت</div>`;

        grid.appendChild(card);

    });

}


function showCategory(catId) {

    const cat = rawData.find(c => c.id === catId);

    if (!cat) return;

    document.getElementById('homeView').classList.add('hidden');

    document.getElementById('detailView').classList.remove('hidden');

    document.getElementById('categoryTitle').innerText = `${cat.icon} ${cat.title}`;

    document.querySelectorAll('.sidebar-item').forEach(b => b.classList.remove('active'));

    const sideBtn = document.getElementById(`side-${catId}`);

    if (sideBtn) sideBtn.classList.add('active');

    renderPrompts(cat.items);

    document.getElementById('scrollArea').scrollTo({ top: 0, behavior: 'smooth' });

}


function renderPrompts(prompts) {

    const list = document.getElementById('promptsList');

    list.innerHTML = '';

    prompts.forEach((p, index) => {

        const box = document.createElement('div');

        box.className = "prompt-box notion-shadow";

        box.innerHTML = `<div class="prompt-content"><span class="text-gray-300 ml-2">#${index+1}</span>${p}</div><button class="copy-btn" onclick="copyText(this)">نسخ</button>`;

        list.appendChild(box);

    });

}


function handleSearch() {

    const query = document.getElementById('globalSearch').value.trim().toLowerCase();

    if (!query) { renderHome(); return; }

    document.getElementById('homeView').classList.add('hidden');

    document.getElementById('detailView').classList.remove('hidden');

    document.getElementById('categoryTitle').innerText = "🔍 نتائج البحث";

    const results = [];

    rawData.forEach(cat => {

        cat.items.forEach(p => {

            if (p.toLowerCase().includes(query)) results.push(p);

        });

    });

    renderPrompts(results);

}


function copyText(btn) {

    const text = btn.previousElementSibling.innerText.replace(/#\d+\s/, '');

    navigator.clipboard.writeText(text).then(() => {

        const toast = document.getElementById('toast');

        toast.style.display = 'block';

        setTimeout(() => { toast.style.display = 'none'; }, 1500);

    });

}


function showHome() {

    document.getElementById('globalSearch').value = '';

    renderHome();

}


window.onload = init;

</script>
